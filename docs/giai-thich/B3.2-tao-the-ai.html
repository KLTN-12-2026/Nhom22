Mục tiêu và luồng

Người học dán đoạn văn tiếng Anh (hoặc tải tệp Word), AI đề xuất thẻ nháp (từ, từ loại, nghĩa, ví dụ, bản dịch), người học chọn thẻ nào lưu và lưu vào bộ có sẵn hoặc bộ mới. Gọi AI chậm và có thể lỗi, nên API trả 202 ngay rồi FE hỏi trạng thái; việc gọi AI chạy trong tác vụ nền của B3.1.

  1. Controller lấy user từ phiên, @Valid body.
  2. CardDraftService kiểm đầu vào, tệp, bộ (ngoài transaction), rồi trong một transaction: QuotaService.giuCho → lưu yeu_cau_ai(CHO, GIU_CHO) → JobService.enqueue("AI_TAO_THE") → 202.
  3. JobWorker (B3.1) lấy việc bằng SKIP LOCKED, gọi CardDraftJobHandler.xuLy không có transaction.
  4. Handler chuyển DANG_XU_LY, đọc đoạn văn hoặc .docx, dựng prompt, gọi AiClient, ghi metadata (TK §14.2), kiểm schema, đánh dấu trùng, ghi thẻ nháp.
  5. Worker chốt trạng thái tác vụ và gọi khiKetThuc trong transaction: HOAN_TAT + xác nhận lượt, LOI + hoàn lượt, hoặc KHONG_XAC_DINH + giữ lượt.
  6. FE hỏi GET /ai/requests/{id} mỗi 2 giây; khi HOAN_TAT có banNhap[].

import org.apache.poi.xwpf.usermodel.XWPFDocument;
import org.springframework.stereotype.Component;

@Component
public class DocxTextReader {

    public static final String MIME_TYPE =
            "application/vnd.openxmlformats-officedocument.wordprocessingml.document";

    public static class UnreadableDocumentException extends RuntimeException {
        public UnreadableDocumentException(String message) {
            super(message);
        }
    }

    public DocxTextReader() {
        ZipSecureFile.setMinInflateRatio(0.01);
        ZipSecureFile.setMaxEntrySize(20L * 1024 * 1024);
        ZipSecureFile.setMaxTextSize(2L * 1024 * 1024);
    }

    public String read(byte[] content) {
        try (var extractor = new XWPFWordExtractor(new XWPFDocument(new ByteArrayInputStream(content)))) {
            return normalize(extractor.getText());
        } catch (IOException | RuntimeException e) {
            throw new UnreadableDocumentException("Không đọc được tệp Word");
        }
    }

    public void ensureReadable(byte[] content) {
        read(content);
    }

    private String normalize(String text) {
        return text.replace("\r\n", "\n")
                .replace('\r', '\n')
                .replaceAll("[\\t\\x0B\\f\\u00A0]+", " ")
                .replaceAll("\n{3,}", "\n\n")
                .strip();
    }

import java.time.Instant;
import java.time.LocalDateTime;
import java.time.ZoneOffset;
import java.util.Map;
import java.util.Optional;
import lombok.RequiredArgsConstructor;
import org.springframework.jdbc.core.namedparam.NamedParameterJdbcTemplate;
import org.springframework.stereotype.Repository;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;

@Repository
@RequiredArgsConstructor
@Transactional(propagation = Propagation.MANDATORY)
public class CardDraftCommitRepository {

    private final NamedParameterJdbcTemplate jdbc;

    public Optional<Entry> find(Long userId, String key) {
        return read(userId, key, false);
    }

    public Entry reserveAndLock(Long userId, String key, Long yeuCauId, String hash, Instant now) {
        jdbc.update("""
                INSERT INTO yeu_cau_luu_the_ai
                (nguoi_dung_id,idempotency_key,yeu_cau_ai_id,request_hash,created_at,updated_at)
                VALUES (:user,:key,:yeuCau,:hash,:now,:now)
                ON DUPLICATE KEY UPDATE id=id
                """, Map.of("user", userId, "key", key, "yeuCau", yeuCauId, "hash", hash,
                "now", utc(now)));
        return read(userId, key, true).orElseThrow(() -> new IllegalStateException("Commit request vanished"));
    }

    public void complete(Long id, Long boTheId, String response, Instant now) {
        int changed = jdbc.update("""
                UPDATE yeu_cau_luu_the_ai
                SET bo_the_id=:deck,response_json=:response,hoan_tat_at=:now,updated_at=:now
                WHERE id=:id AND hoan_tat_at IS NULL
                """, Map.of("id", id, "deck", boTheId, "response", response, "now", utc(now)));
        if (changed != 1) {
            throw new IllegalStateException("AI commit request cannot be completed");
        }
    }

    private Optional<Entry> read(Long userId, String key, boolean locked) {
        return jdbc.query("""
                        SELECT id,yeu_cau_ai_id,request_hash,response_json,hoan_tat_at
                        FROM yeu_cau_luu_the_ai WHERE nguoi_dung_id=:user AND idempotency_key=:key
                        """ + (locked ? " FOR UPDATE" : ""), Map.of("user", userId, "key", key),
                (rs, n) -> new Entry(rs.getLong("id"), rs.getLong("yeu_cau_ai_id"),
                        rs.getString("request_hash"), rs.getString("response_json"),
                        rs.getTimestamp("hoan_tat_at") != null))
                .stream().findFirst();
    }

    private LocalDateTime utc(Instant instant) {
        return LocalDateTime.ofInstant(instant, ZoneOffset.UTC);
    }

    public record Entry(Long id, Long yeuCauId, String hash, String responseJson, boolean completed) {
    }

        body.put("messages", List.of(

                Map.of("role", "user", "content", request.userPrompt())));
        body.put("response_format", Map.of(

                "json_schema", Map.of("name", request.schemaName(), "strict", true, "schema", request.schema())));

            root = json.readTree(raw == null ? "" : raw);
        } catch (JacksonException e) {
            throw new JobException(Kieu.LOI, "SAI_SCHEMA");
        }
        JsonNode message = root.path("choices").path(0).path("message");
        if (message.path("refusal").isString()) {
            throw new JobException(Kieu.LOI, "NCC_TU_CHOI_ND");
        }
        if (!message.path("content").isString()) {
            throw new JobException(Kieu.LOI, "SAI_SCHEMA");

        return new AiResponse(
                message.path("content").asString(),
                root.path("model").isString() ? root.path("model").asString() : props.model(),
                root.path("id").isString() ? root.path("id").asString() : null,

            if (cause instanceof HttpConnectTimeoutException || cause instanceof ConnectException

                return new JobException(Kieu.THU_LAI, "KET_NOI");

            if (cause instanceof HttpTimeoutException || cause instanceof SocketTimeoutException) {
                return new JobException(Kieu.KHONG_XAC_DINH, "TIMEOUT");
            }
        }
        log.warn("OpenAI connection broke: {}", e.getClass().getSimpleName());
        return new JobException(Kieu.KHONG_XAC_DINH, "MAT_KET_NOI");
    }

                || tu.split(" ").length > maxTuMoiThe) {
            return null;
        }

            if (value != null && UNSAFE.matcher(value).find()) {
                return null;
            }
        }
        return new CardDraftData(tu, edit.tuLoai(), edit.nghiaVi(), edit.viDuEn(), edit.dichVi(), false, null);
    }

    private CardDraftData item(JsonNode node, int maxTuMoiThe) {

            return null;
        }

        if (tu == null || tu.isEmpty() || tu.length() > MAX_TU || !TERM.matcher(tu).matches()
                || tu.split(" ").length > maxTuMoiThe) {
            return null;
        }
        String nghiaVi = CardText.term(text(node, "nghiaVi"));
        if (nghiaVi == null || nghiaVi.isEmpty() || nghiaVi.length() > MAX_NGHIA || UNSAFE.matcher(nghiaVi).find()) {
            return null;
        }
        return new CardDraftData(tu, tuLoai(text(node, "tuLoai")), nghiaVi,
                optional(text(node, "viDuEn")), optional(text(node, "dichVi")), false, null);
    }

    private String tuLoai(String value) {
        String normalized = CardText.term(value);
        if (normalized == null) {
            return null;
        }
        normalized = normalized.toLowerCase(Locale.ROOT);
        return normalized.length() <= MAX_TU_LOAI && TU_LOAI.contains(normalized) ? normalized : null;
    }

    private String optional(String value) {
        String normalized = CardText.term(value);
        if (normalized == null || normalized.isEmpty() || normalized.length() > MAX_VI_DU
                || UNSAFE.matcher(normalized).find()) {
            return null;
        }
        return normalized;
    }

    private String text(JsonNode node, String field) {
        JsonNode value = node.path(field);
        return value.isString() ? value.asString() : null;

        boolean existingDeck = request.boTheId() != null;
        boolean newDeck = request.tenBoMoi() != null && !request.tenBoMoi().isBlank();
        if (existingDeck == newDeck) {
            throw new ApiException(ErrorCode.VALIDATION_FAILED, "boTheId",
                    "Chọn bộ thẻ có sẵn hoặc nhập tên bộ mới, không chọn cả hai");
        }
        Map<Long, CardDraftData> edits = edits(request, ids);
        String hash = CardDraftService.sha256(yeuCauId + ":" + ids + ":"
                + (existingDeck ? "deck:" + request.boTheId() : "new:" + request.tenBoMoi())
                + ":" + json.writeValueAsString(new TreeMap<>(edits)));

        var previous = commits.find(userId, key);
        if (previous.isPresent()) {
            return replay(previous.get(), yeuCauId, hash);
        }
        YeuCauAi yeuCau = requests.findOwnedForUpdate(yeuCauId, userId)
                .filter(found -> found.getLoai() == LoaiYeuCauAi.TAO_THE)
                .orElseThrow(() -> new ApiException(ErrorCode.NOT_FOUND, "Không tìm thấy yêu cầu AI"));
        var entry = commits.reserveAndLock(userId, key, yeuCauId, hash, clock.instant());
        if (entry.completed()) {
            return replay(entry, yeuCauId, hash);
        }
        check(entry, yeuCauId, hash);
        if (yeuCau.getTrangThai() != TrangThaiYeuCauAi.HOAN_TAT) {
            throw new ApiException(ErrorCode.BUSINESS_RULE, "Yêu cầu AI chưa tạo xong thẻ nháp");
        }
        List<BanNhapTheAi> selected = drafts.findByYeuCauIdAndIdInOrderByThuTu(yeuCauId, ids);
        if (selected.size() != ids.size()) {
            throw new ApiException(ErrorCode.NOT_FOUND, "banNhapIds", "Không tìm thấy thẻ nháp");
        }
        if (selected.stream().anyMatch(draft -> draft.getTrangThaiDuyet() != TrangThaiDuyet.CHO_DUYET)) {
            throw new ApiException(ErrorCode.CONFLICT, "banNhapIds", "Có thẻ nháp đã được lưu trước đó");
        }

        Long deckId = existingDeck
                ? targets.lockOwnedDeck(userId, Long.valueOf(request.boTheId()))
                : targets.createPrivateDeck(userId, request.tenBoMoi());
        Map<String, List<Long>> index = targets.termIndex(userId, deckId);
        List<ResultItem> results = new ArrayList<>();
        for (BanNhapTheAi draft : selected) {
            CardDraftData data = edits.get(draft.getId());
            if (data == null) {
                data = cardDrafts.read(draft);
            } else {
                draft.suaNoiDung(json.writeValueAsString(data));
            }
            List<Long> existing = index.getOrDefault(CardDraftTargetService.termKey(data.tu()), List.of());
            Long theId = targets.createCard(deckId, data.tu(), data.tuLoai(), data.nghiaVi(), data.viDuEn(), data.dichVi());
            draft.daLuu(theId);
            results.add(new ResultItem(draft.getId().toString(), theId.toString(), !existing.isEmpty(),
                    existing.stream().map(String::valueOf).toList()));
        }
        var response = new CardDraftCommitResponse(yeuCauId.toString(), deckId.toString(), newDeck,
                results.size(), List.copyOf(results));
        commits.complete(entry.id(), deckId, json.writeValueAsString(response), clock.instant());
        return response;
    }

    private Map<Long, CardDraftData> edits(CardDraftCommitRequest request, List<Long> ids) {
        Map<Long, CardDraftData> edits = new HashMap<>();
        for (CardDraftEdit edit : request.sua() == null ? List.<CardDraftEdit>of() : request.sua()) {
            Long id = Long.valueOf(edit.banNhapId());
            if (!ids.contains(id) || edits.containsKey(id)) {
                throw new ApiException(ErrorCode.VALIDATION_FAILED, "sua",
                        "Chỉ sửa thẻ nháp đang chọn, mỗi thẻ một lần");
            }
            CardDraftData data = parser.kiemTraSua(edit, props.cardDraft().maxTuMoiThe());
            if (data == null) {
                throw new ApiException(ErrorCode.VALIDATION_FAILED, "sua",
                        "Từ chỉ gồm chữ cái, khoảng trắng, dấu nháy hoặc gạch nối, tối đa "
                                + props.cardDraft().maxTuMoiThe() + " từ; không dùng ký tự < > { }");
            }
            edits.put(id, data);
        }
        return edits;
    }

    private CardDraftCommitResponse replay(CardDraftCommitRepository.Entry entry, Long yeuCauId, String hash) {
        check(entry, yeuCauId, hash);
        if (!entry.completed() || entry.responseJson() == null) {
            throw new ApiException(ErrorCode.CONFLICT, "Idempotency-Key", "Yêu cầu lưu thẻ đang được xử lý");
        }
        return json.readValue(entry.responseJson(), CardDraftCommitResponse.class);
    }

    private void check(CardDraftCommitRepository.Entry entry, Long yeuCauId, String hash) {
        if (!entry.yeuCauId().equals(yeuCauId) || !entry.hash().equals(hash)) {
            throw new ApiException(ErrorCode.CONFLICT, "Idempotency-Key", "Idempotency-Key đã dùng cho yêu cầu khác");
        }
    }

    private Map<String, Object> with(Map<String, Object> target, Map<String, Object> edit) {
        var body = new java.util.HashMap<>(target);
        body.put("sua", List.of(edit));
        return body;
    }

    private TepTin pending(byte[] content) throws Exception {
        return TepTin.pending(userId, "pending/x", DocxTextReader.MIME_TYPE, content.length, sha256(content),
                LoaiTep.TAI_LIEU);
    }

    private TepTin completed(TepTin file) {
        file.complete(file.getObjectKey(), file.getMimeType(), file.getKichThuoc(), file.getChecksum(),
                clock.instant());
        return file;
    }

    private Long storeDocument(byte[] content) throws Exception {
        String key = "files/" + userId + "/" + UUID.randomUUID();
        storage.putVerified(key, content, DocxTextReader.MIME_TYPE);
        return files.saveAndFlush(completed(TepTin.pending(userId, key, DocxTextReader.MIME_TYPE, content.length,
                sha256(content), LoaiTep.TAI_LIEU))).getId();
    }

    private byte[] docx(String... paragraphs) throws Exception {
        try (var document = new XWPFDocument(); var out = new ByteArrayOutputStream()) {
            for (String paragraph : paragraphs) {
                document.createParagraph().createRun().setText(paragraph);
            }
            document.write(out);
            return out.toByteArray();
        }
    }

    private String sha256(byte[] content) throws Exception {
        return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(content));
    }
