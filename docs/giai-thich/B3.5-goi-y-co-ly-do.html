Mục tiêu và luồng

GET /api/v1/learning/recommendations trả mảng [{reasonCode, message, cardIds, evidence}]: mỗi phần tử là một lời khuyên kèm lý do có số liệu chứng minh (TK §8.3). User chưa có dữ liệu → []. Không ghi dữ liệu, không migration.

RecommendationController→RecommendationService→5 RecommendationRule→SoTayTuKhoRepository / LearningPlanService→MySQL
  1. Controller lấy userId từ phiên đăng nhập.
  2. Service đọc Clock một lần, sắp rule theo thứ tự enum MaLyDoGoiY.
  3. Mỗi rule truy vấn tổng hợp, lọc thẻ dùng được (của chính user, chưa xóa, không tạm ngưng), sinh ứng viên đã sắp theo độ mạnh.
  4. Service lấy tối đa 2 ứng viên mỗi lý do, tối đa 10 tổng, bỏ ứng viên có thẻ đã thuộc lý do ưu tiên cao hơn.
  5. Trả JSON; evidence bỏ trường null.

application.yamlSửa — thêm khối app.recommendations

backend/k28/src/main/resources/application.yaml

Nơi đặt mọi ngưỡng của B3.5. Không có số nào ghi cứng trong Java.

  • toi-da-goi-y: 10 — trần tổng số gợi ý trả về một lần gọi API.
  • toi-da-moi-ly-do: 2 — mỗi mã lý do được tối đa 2 gợi ý. Với 5 mã đang bật: 2 (QUEN_NHIEU) + 2 (SAI_CHINH_TA) + 2 (CAP_DE_NHAM) + 1 (QUA_HAN_NHIEU chỉ sinh 1) + 2 (TIEN_BO) = 9 ≤ 10, nên thẻ khen TIEN_BO (R-04) không bao giờ bị cắt vì hết chỗ.
  • toi-da-the-qua-han: 20 — số thẻ quá hạn lâu nhất đưa vào cardIds của QUA_HAN_NHIEU, để response không phình khi user nợ hàng trăm thẻ.
  • tien-bo.so-ngay-cu: 30 / so-ngay-moi: 7 — hai cửa sổ thời gian: cửa sổ cũ [now−30 ngày, now−7 ngày), cửa sổ mới [now−7 ngày, now). Hai cửa sổ nối tiếp, không chồng nhau.
  • so-luot-cu-toi-thieu: 5, so-sai-cu-toi-thieu: 3, ty-le-sai-cu-toi-thieu: 0.5 — điều kiện “từng sai nhiều”: đủ 5 lượt quan sát, ít nhất 3 lượt sai và tỷ lệ sai ≥ 50%. Có tỷ lệ để thẻ sai 3/20 lượt không bị coi là “sai nhiều”.
  • so-luot-moi-toi-thieu: 5, so-sai-moi-toi-da: 0 — điều kiện “gần đây đúng”: đủ 5 lượt trong 7 ngày và không sai lượt nào. Đây là ngưỡng “đủ lượt quan sát ở cả hai cửa sổ” của R-04.
  • phat-am-thap.enabled: false — cờ tắt PHAT_AM_THAP. Chỉ bật khi B3.4b dùng dịch vụ phát âm thật (không bật khi đang dùng Fake “Mô phỏng”). diem-toi-thieu: 60 là ngưỡng điểm dành sẵn cho rule đó.
  • Ba mã QUEN_NHIEU, SAI_CHINH_TA, CAP_DE_NHAM không có khóa riêng ở đây: chúng dùng chung app.notebook.* của sổ tay B2.9 để hai màn UI27 (sổ tay) và UI28 (gợi ý) không lệch nhau.
  • Spring Boot đọc tên kebab-case (so-ngay-cu) vào tham số camelCase (soNgayCu) theo cơ chế relaxed binding.

package com.do_an_tot_nghiep.k28.practice.service.recommendation;

import jakarta.validation.Valid;
import jakarta.validation.constraints.DecimalMax;
import jakarta.validation.constraints.DecimalMin;
import jakarta.validation.constraints.Max;
import jakarta.validation.constraints.Min;
import jakarta.validation.constraints.NotNull;
import org.springframework.boot.context.properties.ConfigurationProperties;
import org.springframework.validation.annotation.Validated;

@Validated
@ConfigurationProperties("app.recommendations")
public record RecommendationProperties(
        @Min(1) int toiDaGoiY,
        @Min(1) int toiDaMoiLyDo,
        @Min(1) int toiDaTheQuaHan,
        @NotNull @Valid TienBo tienBo,
        @NotNull @Valid PhatAmThap phatAmThap
) {

    public record TienBo(
            @Min(2) int soNgayCu,
            @Min(1) int soNgayMoi,
            @Min(1) int soLuotCuToiThieu,
            @Min(1) int soSaiCuToiThieu,
            @DecimalMin("0.0") @DecimalMax("1.0") double tyLeSaiCuToiThieu,
            @Min(1) int soLuotMoiToiThieu,
            @Min(0) int soSaiMoiToiDa
    ) {

        public TienBo {
            if (soNgayCu <= soNgayMoi) {
                throw new IllegalArgumentException(
                        "app.recommendations.tien-bo.so-ngay-cu phải lớn hơn so-ngay-moi");
            }
        }
    }

    public record PhatAmThap(boolean enabled, @Min(0) @Max(100) int diemToiThieu) {
    }
}

package com.do_an_tot_nghiep.k28.practice.service.recommendation;

import java.time.Duration;
import java.time.Instant;

public record ProgressPolicy(RecommendationProperties.TienBo config) {

    public Instant oldFrom(Instant now) {
        return now.minus(Duration.ofDays(config.soNgayCu()));
    }

    public Instant recentFrom(Instant now) {
        return now.minus(Duration.ofDays(config.soNgayMoi()));
    }

    public boolean matches(long soLuotCu, long soSaiCu, long soLuotMoi, long soSaiMoi) {
        return soLuotCu >= config.soLuotCuToiThieu()
                && soSaiCu >= config.soSaiCuToiThieu()
                && soSaiCu >= config.tyLeSaiCuToiThieu() * soLuotCu
                && soLuotMoi >= config.soLuotMoiToiThieu()
                && soSaiMoi <= config.soSaiMoiToiDa();
    }
}

                .filter(pair -> pair.getSoLanNham() >= properties.capNham())

            ids.add(pair.getThe2Id());
        });

        return pairs.stream()
                .filter(pair -> usable.containsKey(pair.getThe1Id()) && usable.containsKey(pair.getThe2Id()))
                .filter(pair -> visible(hidden, pair.getThe1Id(), pair) && visible(hidden, pair.getThe2Id(), pair))
                .map(pair -> new RecommendationCandidate(
                        code(),
                        "Phân biệt ‘%s’ và ‘%s’: bạn chọn nhầm %d lần".formatted(

                                .soLanNham(pair.getSoLanNham())
                                .lanCuoiAt(pair.getLanCuoiAt())
                                .build(),
                        true))
                .toList();
    }

    private static boolean visible(Map<Long, Instant> hidden, Long theId, PairStatRow pair) {

package com.do_an_tot_nghiep.k28.practice.service.recommendation;

import com.do_an_tot_nghiep.k28.practice.dto.RecommendationResponse;
import java.time.Clock;
import java.time.Instant;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@RequiredArgsConstructor
public class RecommendationService {

    private final List<RecommendationRule> rules;
    private final RecommendationProperties properties;
    private final Clock clock;

    @Transactional
    public List<RecommendationResponse> list(Long userId) {
        Instant now = clock.instant();
        Set<Long> used = new HashSet<>();
        List<RecommendationResponse> result = new ArrayList<>();
        List<RecommendationRule> ordered = rules.stream()
                .sorted(Comparator.comparing(RecommendationRule::code))
                .toList();
        for (RecommendationRule rule : ordered) {
            if (result.size() >= properties.toiDaGoiY()) {
                break;
            }
            int taken = 0;
            for (RecommendationCandidate candidate : rule.evaluate(userId, now)) {
                if (taken >= properties.toiDaMoiLyDo() || result.size() >= properties.toiDaGoiY()) {
                    break;
                }
                if (candidate.exclusive()) {
                    if (candidate.cardIds().stream().anyMatch(used::contains)) {
                        continue;
                    }
                    used.addAll(candidate.cardIds());
                }
                result.add(candidate.toResponse());
                taken++;
            }
        }
        return result;
    }
}
